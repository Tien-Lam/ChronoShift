async function seedShare(key:string) {
  await seed.evaluate(async key=>{
    await new Promise<void>((resolve,reject)=>{
      const r=indexedDB.open('chronoshift-handoff',1);
      r.onupgradeneeded=()=>r.result.createObjectStore('messages');
      r.onerror=()=>reject(r.error);
      r.onsuccess=()=>{const db=r.result,tx=db.transaction('messages','readwrite');
        tx.objectStore('messages').put({text:'April 10, 2026 8pm UTC',created:Date.now()},key);
        tx.oncomplete=()=>{db.close();resolve();};
      };
    });
  },key);
}

  const open=indexedDB.open.bind(indexedDB);
  indexedDB.open=((...args:Parameters<typeof indexedDB.open>)=>{
    const r=open(...args);if(args[0]!=='chronoshift-handoff')return r;
    return new Proxy(r,{get:(t,k)=>Reflect.get(t,k,t),set(t,k,fn){
      if(k==='onsuccess'){t.onsuccess=e=>{(window as any).releaseShare=()=>fn.call(t,e);};return true;}
      return Reflect.set(t,k,fn,t);
    }});
  }) as typeof indexedDB.open;
  const NativeWorker=Worker;
  (window as any).Worker=class extends NativeWorker {
    set onmessage(fn:any) {super.onmessage=(event:MessageEvent)=>{
      if((window as any).holdWorker)(window as any).releaseWorker=()=>fn(event);else fn(event);
    };}
