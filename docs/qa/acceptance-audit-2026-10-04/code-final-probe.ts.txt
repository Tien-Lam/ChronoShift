const page = await context.newPage();
const errors: string[] = [];

const evidence: any = { revision: '49207b81cc013c3c411a52de2a710f6af7b3705c', base: 'b432db067ad8176e7f040c74997e2021b629a07c', browser: browser.version(), profile: 'isolated Chromium desktop, en-AU, Australia/Sydney', cases: [] };
await page.addInitScript(() => {
  (window as any).workerDeliveries = [];
  const descriptor = Object.getOwnPropertyDescriptor(Worker.prototype, 'onmessage')!;
  Object.defineProperty(Worker.prototype, 'onmessage', {
    configurable: true,
    get: descriptor.get,
    set(callback) { descriptor.set!.call(this, (event: MessageEvent) => (window as any).workerDeliveries.push(() => callback.call(this, event))); }
  });
  (window as any).pastes = []; (window as any).copies = [];
  Object.defineProperty(navigator, 'clipboard', { value: {
    readText: () => new Promise((resolve, reject) => (window as any).pastes.push({resolve, reject})),
    writeText: (value: string) => new Promise((resolve, reject) => (window as any).copies.push({resolve, reject, value})),
  }});
});

const convert = page.getByRole('button', {name:'Convert', exact:true});

const held = async () => expect.poll(() => page.evaluate(() => (window as any).workerDeliveries.length)).toBe(1);
